DO $rollback$
DECLARE saved record;grant_value jsonb;p regprocedure;grantee text;
BEGIN
 LOCK TABLE public.zasp_security_agent_runs,public.zasp_security_agent_definitions,public.zasp_security_agent_definition_versions,public.zasp_security_agent_plans,public.zasp_security_agent_steps,public.zasp_security_agent_kill_switches,public.zasp_security_agent_request_receipts,public.zasp_security_agent_effects,public.zasp_compliance_export_jobs,public.zasp_sa_export_links IN ACCESS EXCLUSIVE MODE NOWAIT;
 IF EXISTS(SELECT 1 FROM public.zasp_sa_export_links)
 OR EXISTS(SELECT 1 FROM public.zasp_compliance_export_jobs WHERE job_origin='agent_run')
 OR EXISTS(SELECT 1 FROM public.zasp_security_agent_definitions WHERE body->'allowed_actions' ? 'create_evidence_export')
 OR EXISTS(SELECT 1 FROM public.zasp_security_agent_definition_versions WHERE definition->'allowed_actions' ? 'create_evidence_export')
 OR EXISTS(SELECT 1 FROM public.zasp_security_agent_plans WHERE plan->'verification'->>'kind'='export')
 OR EXISTS(SELECT 1 FROM public.zasp_security_agent_steps WHERE action_key='create_evidence_export')
 OR EXISTS(SELECT 1 FROM public.zasp_security_agent_effects WHERE action_key='create_evidence_export')
 OR EXISTS(SELECT 1 FROM public.zasp_security_agent_kill_switches WHERE action_key='create_evidence_export')
 OR EXISTS(SELECT 1 FROM public.zasp_security_agent_request_receipts WHERE operation='setSecurityAgentExecutionControl' AND resource_id='action:create_evidence_export') THEN RAISE EXCEPTION USING ERRCODE='55000',MESSAGE='export history retained';END IF;
 ALTER TABLE public.zasp_compliance_export_jobs DROP CONSTRAINT zasp_sa_export_parent;
 DROP TABLE public.zasp_sa_export_links;
 ALTER TABLE public.zasp_compliance_export_jobs DROP CONSTRAINT zasp_sa_export_origin,DROP CONSTRAINT zasp_sa_export_idempotency,DROP CONSTRAINT zasp_sa_export_job_binding;
 ALTER TABLE public.zasp_compliance_export_jobs DROP COLUMN job_origin,DROP COLUMN agent_run_id,DROP COLUMN agent_step_id,ALTER COLUMN session_digest SET NOT NULL;
 FOR saved IN SELECT * FROM zasp_sa_export_prior.job_constraints ORDER BY name LOOP
  EXECUTE format('ALTER TABLE public.zasp_compliance_export_jobs ADD CONSTRAINT %I %s',saved.name,saved.definition);
 END LOOP;
 FOR saved IN SELECT * FROM zasp_sa_export_prior.functions ORDER BY signature LOOP
  EXECUTE saved.definition;
  EXECUTE format('ALTER FUNCTION %s OWNER TO %I',saved.signature,saved.owner_name);
  FOR grantee IN SELECT CASE WHEN a.grantee=0 THEN 'PUBLIC' ELSE a.grantee::regrole::text END FROM pg_proc p CROSS JOIN LATERAL aclexplode(COALESCE(p.proacl,acldefault('f',p.proowner))) a WHERE p.oid=saved.signature::regprocedure LOOP
   EXECUTE format('REVOKE ALL ON FUNCTION %s FROM %s',saved.signature,CASE WHEN grantee='PUBLIC' THEN 'PUBLIC' ELSE quote_ident(grantee) END);
  END LOOP;
  FOR grant_value IN SELECT value FROM jsonb_array_elements(saved.acl) LOOP
   EXECUTE format('GRANT EXECUTE ON FUNCTION %s TO %s%s',saved.signature,CASE WHEN grant_value->>'grantee'='PUBLIC' THEN 'PUBLIC' ELSE quote_ident(grant_value->>'grantee') END,CASE WHEN (grant_value->>'grantable')::boolean THEN ' WITH GRANT OPTION' ELSE '' END);
  END LOOP;
 END LOOP;
 FOR p IN SELECT oid::regprocedure FROM pg_proc WHERE pronamespace='public'::regnamespace AND starts_with(proname,'zasp_sa_export_') LOOP EXECUTE format('DROP FUNCTION %s',p);END LOOP;
END $rollback$;
DROP SCHEMA zasp_sa_export_prior CASCADE;
DELETE FROM public.zasp_schema_metadata WHERE key IN('production_security_agent_exports_checksum','production_security_agent_exports_fingerprint');
