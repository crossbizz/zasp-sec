func TestSecurityAgentExportPlannerAdmissionPostgres(t *testing.T) {
	for _, mode := range []string{"autonomous", "supervised", "failure"} {
		t.Run(mode, func(t *testing.T) {
			activation := mode
			if mode == "failure" {
				activation = "autonomous"
			}
			runExportPlannerFixture(t, activation, func(f *exportDBFixture) {
				var count int
				if err := f.owner.QueryRow(f.ctx, `SELECT (SELECT count(*) FROM zasp_security_agent_plans WHERE run_id=$1)+(SELECT count(*) FROM zasp_security_agent_steps WHERE run_id=$1)+(SELECT count(*) FROM zasp_security_agent_provider_reservations WHERE run_id=$1)`, exportFixtureRun).Scan(&count); err != nil || count != 0 {
					t.Fatalf("seeded execution authority: %d %v", count, err)
				}
				raw, err := exportPlannerContext(f)
				if err != nil {
					t.Fatal(err)
				}
				var envelope struct {
					Context struct {
						ExportSelection []SecurityAgentExportSelection `json:"export_selection"`
						AllowedTargets  []string                       `json:"allowed_targets"`
					} `json:"context"`
					InputDigest string `json:"input_digest"`
				}
				if json.Unmarshal(raw, &envelope) != nil || len(envelope.Context.ExportSelection) != 2 || envelope.Context.ExportSelection[0].Kind != "finding" || envelope.Context.ExportSelection[0].ID != exportFixtureFinding || envelope.Context.ExportSelection[1].Kind != "run_audit" || !reflect.DeepEqual(envelope.Context.AllowedTargets, []string{exportFixtureRun}) {
					t.Fatalf("derived context=%s", raw)
				}
				// A separate transaction appends a later audit before reservation. It must
				// never expand the already displayed model input or change its digest.
				if _, err = f.owner.Exec(f.ctx, `INSERT INTO zasp_security_agent_audit(organization_id,workspace_id,environment_id,audit_id,correlation_id,run_id,actor_id,event_kind,event_digest,body) VALUES($1,$2,$3,'pid_8e150003-0000-4000-8000-000000000003','pid_8e150004-0000-4000-8000-000000000004',$4,$5,'later_audit',digest('later audit','sha256'),'{}')`, f.o, f.w, f.e, exportFixtureRun, f.actor); err != nil {
					t.Fatal(err)
				}
				again, err := exportPlannerContext(f)
				if err != nil || string(again) != string(raw) {
					t.Fatalf("later audit changed original input: %s %v", again, err)
				}
				input := envelope.InputDigest[7:]
				output := strings.Repeat("ab", 32)
				var attempt int
				if err = f.owner.QueryRow(f.ctx, `SELECT attempt FROM zasp_security_agent_runs WHERE run_id=$1`, exportFixtureRun).Scan(&attempt); err != nil {
					t.Fatal(err)
				}
				if err = f.worker.QueryRow(f.ctx, `SELECT zasp_sa_export_reserve_planner($1,$2,$3,$4,$5,$6,$7,'export-planner-reservation',decode($8,'hex'),'fixture-model','fixture-cost-policy','openrouter_credit',50,100)`, f.o, f.w, f.e, exportFixtureRun, exportFixtureWorker, exportFixtureLease, attempt, input).Scan(&raw); err != nil {
					t.Fatal(err)
				}
				if !strings.Contains(string(raw), `"budget_permit"`) {
					t.Fatalf("no reservation: %s", raw)
				}
				if err = f.worker.QueryRow(f.ctx, `SELECT zasp_security_agent_budget_settle_planner($1,$2,$3,$4,$5,$6,$7,'export-planner-reservation',decode($8,'hex'),10,10,20,20)`, f.o, f.w, f.e, exportFixtureRun, exportFixtureWorker, exportFixtureLease, attempt, output).Scan(&raw); err != nil {
					t.Fatal(err)
				}
				selected, _ := json.Marshal(envelope.Context.ExportSelection[:1])
				f.selection = selected
				candidate, _ := json.Marshal(map[string]any{"version": 1, "summary": "Export selected run evidence", "steps": []any{map[string]any{"index": 0, "action": "create_evidence_export", "target_id": exportFixtureRun, "evidence_ids": json.RawMessage(selected)}}})
				invoke := func() (json.RawMessage, error) {
					var result json.RawMessage
					var err error
					if mode == "failure" {
						err = f.worker.QueryRow(f.ctx, `SELECT zasp_sa_export_fail_planner($1,$2,$3,$4,$5,$6,decode($7,'hex'),decode($8,'hex'),'fixture-model','planner-v1','planner_rejected','pid_8e150005-0000-4000-8000-000000000005','pid_8e150006-0000-4000-8000-000000000006')`, f.o, f.w, f.e, exportFixtureRun, exportFixtureWorker, exportFixtureLease, input, output).Scan(&result)
					} else {
						err = f.worker.QueryRow(f.ctx, `SELECT zasp_sa_export_accept_planner($1,$2,$3,$4,$5,$6,decode($7,'hex'),decode($8,'hex'),'fixture-model','planner-v1',$9,$10,$11,'pid_8e150005-0000-4000-8000-000000000005','pid_8e150006-0000-4000-8000-000000000006')`, f.o, f.w, f.e, exportFixtureRun, exportFixtureWorker, exportFixtureLease, input, output, candidate, exportApprovalID, time.Now().UTC().Add(5*time.Minute)).Scan(&result)
					}
					return result, err
				}
				raw, err = invoke()
				if err != nil {
					t.Fatal(err)
				}
				var receipt map[string]any
				if json.Unmarshal(raw, &receipt) != nil {
					t.Fatal(string(raw))
				}
				replay, err := invoke()
				var got map[string]any
				if err != nil || json.Unmarshal(replay, &got) != nil {
					t.Fatalf("lost reply: %s %v", replay, err)
				}
				receipt["replayed"] = true
				if !reflect.DeepEqual(receipt, got) {
					t.Fatalf("replay changed: %s", replay)
				}
				originalInput := input
				input = strings.Repeat("cd", 32)
				if _, err = invoke(); err == nil {
					t.Fatal("changed export request replayed")
				}
				input = originalInput
				if mode == "failure" {
					if receipt["state"] != "failed" {
						t.Fatal(receipt)
					}
					return
				}
				if mode == "supervised" {
					if receipt["state"] != "waiting_approval" {
						t.Fatal(receipt)
					}
					if _, err = f.owner.Exec(f.ctx, `INSERT INTO zasp_identity_memberships(principal_id,organization_id,organization_reference,member_reference,role,active) VALUES($4,$1,'planner-approver-org','planner-approver','security_engineer',true); INSERT INTO zasp_authorized_scopes(principal_id,organization_id,workspace_id,environment_id,label,permissions) VALUES($4,$1,$2,$3,'Planner approver','["view","manage_workflows"]')`, pgx.QueryExecModeSimpleProtocol, f.o, f.w, f.e, exportApproverID); err != nil {
						t.Fatal(err)
					}
					if raw, err = readExportApproval(f); err != nil {
						t.Fatal(err)
					}
					if _, err = decodeApprovalContextEnvelope(raw, exportApprovalID); err != nil {
						t.Fatal(err)
					}
					if _, err = decideExportApproval(f, exportApproverID, "export-planner-approval", time.Now().UTC()); err != nil {
						t.Fatal(err)
					}
				}
				if err = f.worker.QueryRow(f.ctx, postgresSecurityAgentClaimRunsV24SQL, exportFixtureWorker, exportFixtureLease, 120, 1).Scan(&raw); err != nil {
					t.Fatal(err)
				}
				if _, err = f.dispatch(f.o, f.w, f.e, exportFixtureRun, exportFixtureLease); err != nil {
					t.Fatalf("real planner plan did not dispatch: %v", err)
				}
			})
		})
	}
}

// Reconstructing selection from snapshot order must fail this registered case.

	runExportPlannerFixture(t, "autonomous", func(f *exportDBFixture) {
		if _, err := f.owner.Exec(f.ctx, `INSERT INTO zasp_security_agent_audit(organization_id,workspace_id,environment_id,audit_id,correlation_id,run_id,actor_id,event_kind,event_digest,body) VALUES($1,$2,$3,'pid_8e150003-0000-4000-8000-000000000003','pid_8e150004-0000-4000-8000-000000000004',$4,$5,'second_audit',digest('second audit','sha256'),'{}')`, f.o, f.w, f.e, exportFixtureRun, f.actor); err != nil {
			t.Fatal(err)
		}
		raw, err := exportPlannerContext(f)
		if err != nil {
			t.Fatal(err)
		}
		var envelope struct {
			Context struct {
				Selection []SecurityAgentExportSelection `json:"export_selection"`
			} `json:"context"`
			InputDigest string `json:"input_digest"`
		}
		if json.Unmarshal(raw, &envelope) != nil || len(envelope.Context.Selection) != 3 {
			t.Fatalf("expected finding and two audits: %s", raw)
		}
		selected := []SecurityAgentExportSelection{envelope.Context.Selection[2], envelope.Context.Selection[0]}
		input, _ := decodeSecurityAgentDigest(envelope.InputDigest)
		output := []byte(strings.Repeat("x", 32))
		var attempt int
		if err = f.owner.QueryRow(f.ctx, `SELECT attempt FROM zasp_security_agent_runs WHERE run_id=$1`, exportFixtureRun).Scan(&attempt); err != nil {
			t.Fatal(err)
		}

			t.Fatal(err)
		}
		if err = f.worker.QueryRow(f.ctx, `SELECT zasp_security_agent_budget_settle_planner($1,$2,$3,$4,$5,$6,$7,'candidate-order-reservation',$8,10,10,20,20)`, f.o, f.w, f.e, exportFixtureRun, exportFixtureWorker, exportFixtureLease, attempt, output).Scan(&raw); err != nil {
			t.Fatal(err)
		}
		expires := time.Now().UTC().Add(5 * time.Minute)
		invoke := func(refs []SecurityAgentExportSelection) error {
			candidate, marshalErr := json.Marshal(map[string]any{"version": 1, "summary": "Candidate ordered subset", "steps": []any{map[string]any{"index": 0, "action": "create_evidence_export", "target_id": exportFixtureRun, "evidence_ids": refs}}})
			if marshalErr != nil {
				t.Fatal(marshalErr)
			}
			return f.worker.QueryRow(f.ctx, postgresSecurityAgentExportAcceptPlannerSQL, f.o, f.w, f.e, exportFixtureRun, exportFixtureWorker, exportFixtureLease, input, output, "fixture-model", "planner-v1", candidate, exportApprovalID, expires, "pid_8e150005-0000-4000-8000-000000000005", "pid_8e150006-0000-4000-8000-000000000006").Scan(&raw)
		}
		if err = invoke(selected); err != nil {
			t.Fatalf("valid reordered subset refused: %v", err)
		}
		var receipt map[string]any
		if json.Unmarshal(raw, &receipt) != nil || receipt["state"] != "queued" {
			t.Fatalf("acceptance receipt: %s", raw)
		}
		var stored json.RawMessage
		var leaseCleared bool
		if err = f.owner.QueryRow(f.ctx, `SELECT p.plan->'steps'->0->'evidence_ids', r.lease_token IS NULL FROM zasp_security_agent_plans p JOIN zasp_security_agent_runs r USING(organization_id,workspace_id,environment_id,run_id) WHERE p.run_id=$1`, exportFixtureRun).Scan(&stored, &leaseCleared); err != nil {
			t.Fatal(err)
		}

		if json.Unmarshal(stored, &storedRefs) != nil || !reflect.DeepEqual(storedRefs, selected) || !leaseCleared {
			t.Fatalf("candidate order or cleared lease lost: %s %v", stored, leaseCleared)
		}
		if err = invoke(selected); err != nil {
			t.Fatalf("lost reply refused: %v", err)
		}

		receipt["replayed"] = true
		if json.Unmarshal(raw, &replay) != nil || !reflect.DeepEqual(receipt, replay) {
			t.Fatalf("receipt changed: %s", raw)
		}

			t.Fatal("changed candidate subset replayed")
		}
		if got := existingTestAcceptanceSnapshot(t, f.ctx, f.owner, f.o, f.w, f.e, exportFixtureRun); got != before {
			t.Fatal("changed replay persisted authority")
		}
	})

func TestSecurityAgentExportPlannerRefusalsPostgres(t *testing.T) {
	runExportPlannerFixture(t, "autonomous", func(f *exportDBFixture) {
		raw, err := exportPlannerContext(f)
		if err != nil {
			t.Fatal(err)
		}
		var envelope struct {
			Context struct {
				Selection json.RawMessage `json:"export_selection"`
			} `json:"context"`
			InputDigest string `json:"input_digest"`
		}
		if json.Unmarshal(raw, &envelope) != nil {
			t.Fatal(string(raw))
		}
		for _, change := range []struct{ set, restore string }{
			{`UPDATE zasp_risk_findings SET version=2 WHERE id=$1`, `UPDATE zasp_risk_findings SET version=1 WHERE id=$1`},
			{`UPDATE zasp_identity_memberships SET role='security_engineer' WHERE principal_id=$2`, `UPDATE zasp_identity_memberships SET role='security_admin' WHERE principal_id=$2`},
			{`UPDATE zasp_identity_memberships SET active=false WHERE principal_id=$2`, `UPDATE zasp_identity_memberships SET active=true WHERE principal_id=$2`},
		} {
			args := []any{exportFixtureFinding}
			if strings.Contains(change.set, "$2") {
				args = []any{f.actor}
				change.set = strings.ReplaceAll(change.set, "$2", "$1")
				change.restore = strings.ReplaceAll(change.restore, "$2", "$1")
			}
			if _, err = f.owner.Exec(f.ctx, change.set, args...); err != nil {
				t.Fatal(err)
			}
			if _, err = exportPlannerContext(f); err == nil {
				t.Fatalf("stale authority accepted: %s", change.set)
			}
			if _, err = f.owner.Exec(f.ctx, change.restore, args...); err != nil {
				t.Fatal(err)
			}
		}
		var refs []map[string]any
		var originalDefinition json.RawMessage
		if err = f.owner.QueryRow(f.ctx, `SELECT body FROM zasp_security_agent_definitions WHERE organization_id=$1`, f.o).Scan(&originalDefinition); err != nil {

			t.Fatal(err)
		}

		}
		if _, err = f.owner.Exec(f.ctx, `UPDATE zasp_security_agent_definitions SET body=$2::jsonb WHERE organization_id=$1; UPDATE zasp_security_agent_definition_versions SET definition=$2::jsonb,definition_digest=digest(convert_to(($2::jsonb)::text,'UTF8'),'sha256') WHERE organization_id=$1`, pgx.QueryExecModeSimpleProtocol, f.o, originalDefinition); err != nil {

		}
	})
